PROGRAM 3: Counter method: factor of a number
========================================================================

#include <stdio.h> 
int main()
{
   int n, i, counter = 0;
   scanf("%d", &n);
   counter++;
   for (i = 1; i <= n; ++i)
   {
       counter++;
       if (n % i == 0)
       {
           counter++;
       }

       counter++;
   }
   printf("%d", counter);
   return 0;
}

PROGRAM 5: Counter method: reverse a number
========================================================================

#include <stdio.h> 
int main()

{
   int n;
   int rev = 0, remainder;
   int count = 0;
   scanf("%d", &n);
   count += 3;
   while (n != 0)
   {
       count++;
       remainder = n % 10;
       count++;
       rev = rev * 10 + remainder;
       count++;
       n /= 10;
       count++;
   }
   printf("%d", count);
   return 0;
}
